# Canvas UX Enhancements: Implementation Plan

## Task 1: Keyboard shortcuts composable

**File**: `frontend/src/lib/useCanvasShortcuts.ts` (new)

```ts
import { onMounted, onUnmounted } from 'vue'
import { useWorkflowStore } from '@/stores/workflow'

export function useCanvasShortcuts() {
  const workflow = useWorkflowStore()

  function handler(e: KeyboardEvent) {
    const isInput = (e.target as HTMLElement)?.tagName === 'INPUT' ||
                    (e.target as HTMLElement)?.tagName === 'TEXTAREA' ||
                    (e.target as HTMLElement)?.isContentEditable
    if (isInput) return

    const ctrl = e.ctrlKey || e.metaKey

    if (e.key === 'Delete' || e.key === 'Backspace') {
      if (workflow.selectedNodeId) {
        workflow.removeNode(workflow.selectedNodeId)
        e.preventDefault()
      }
    } else if (ctrl && e.key === 'c') {
      if (workflow.selectedNodeId) {
        const node = workflow.nodes.find(n => n.id === workflow.selectedNodeId)
        if (node) workflow.copyNodes([node.id])
      }
      e.preventDefault()
    } else if (ctrl && e.key === 'v') {
      if (workflow.clipboard.length > 0) {
        workflow.pasteNodes()
      }
      e.preventDefault()
    } else if (ctrl && e.key === 'a') {
      workflow.selectAllNodes()
      e.preventDefault()
    } else if (ctrl && e.key === 'z') {
      if (e.shiftKey) workflow.redo()
      else workflow.undo()
      e.preventDefault()
    } else if (ctrl && e.key === 'd') {
      if (workflow.selectedNodeId) {
        workflow.cloneNode(workflow.selectedNodeId)
      }
      e.preventDefault()
    } else if (ctrl && e.key === 'Enter') {
      workflow.runWorkflow()
      e.preventDefault()
    } else if (e.key === 'Escape') {
      workflow.selectedNodeId = null
      e.preventDefault()
    }
  }

  onMounted(() => window.addEventListener('keydown', handler))
  onUnmounted(() => window.removeEventListener('keydown', handler))
}
```

Add to store methods:
```ts
function copyNodes(nodeIds: string[]) {
  clipboard.value = nodes.value
    .filter(n => nodeIds.includes(n.id))
    .map(n => JSON.parse(JSON.stringify(n)))
}

function pasteNodes() {
  pushHistory()
  const offset = 30
  for (const node of clipboard.value) {
    const newId = `${node.data.nodeType}-${Date.now()}-${Math.random().toString(36).slice(2, 4)}`
    ;(nodes.value as VFNode[]).push({
      ...JSON.parse(JSON.stringify(node)),
      id: newId,
      position: { x: node.position.x + offset, y: node.position.y + offset },
      data: { ...node.data, status: 'idle' },
    })
  }
}

function selectAllNodes() {
  nodes.value.forEach(n => { n.selected = true })
}
```

## Task 3: Edge flow animation

**File**: `frontend/src/workflow/canvas/WorkflowCanvas.vue`

Add computed property:
```ts
const edgeStyles = computed(() => {
  const styles: Record<string, any> = {}
  for (const edge of workflow.edges) {
    const isActive = workflow.executionStatus === 'running'
    const srcStatus = workflow.nodeStatuses.get(edge.source)?.status
    const tgtStatus = workflow.nodeStatuses.get(edge.target)?.status
    if (isActive && tgtStatus === 'running') {
      styles[edge.id || ''] = { stroke: '#58a6ff', strokeDasharray: '5 3', animation: 'flow 0.5s linear infinite' }
    } else if (tgtStatus === 'success' || srcStatus === 'success') {
      styles[edge.id || ''] = { stroke: '#3fb950', strokeWidth: 2 }
    } else if (tgtStatus === 'failed') {
      styles[edge.id || ''] = { stroke: '#f85149', strokeWidth: 2 }
    } else {
      styles[edge.id || ''] = { stroke: '#30363d', strokeWidth: 2 }
    }
  }
  return styles
})
```

Bind to template:
```vue
<VueFlow ... :edge-styles="edgeStyles" />
```

Add CSS:
```css

  0% { stroke-dashoffset: 12; }
  100% { stroke-dashoffset: 0; }
}
```
